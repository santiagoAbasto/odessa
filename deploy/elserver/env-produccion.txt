# =====================================================================
# .env de PRODUCCIÓN — sistema.odessaoil.com  (ElServer / LatinCloud)
# =====================================================================
#
# Va en la raíz del sitio, al lado de `artisan`:
#
#     <home>/sistema.odessaoil.com/.env
#
# NO dentro de `public/`: ahí sería descargable por HTTP.
#
# Los valores sensibles están vacíos: se completan una sola vez en el
# servidor y no vuelven a copiarse a ningún archivo del repositorio,
# ni a un chat, ni a una captura.
#
# Permisos una vez cargado:  chmod 600 .env
# =====================================================================

APP_NAME="Odessa Oil"
APP_ENV=production
# Generar EN EL SERVIDOR con:  php artisan key:generate
APP_KEY=
APP_DEBUG=false
APP_URL=https://sistema.odessaoil.com

# La operación es argentina y las reglas comparan contra `today` y `now()`.
# Con UTC, a partir de las 21:00 hora local el servidor ya está en el día
# siguiente y rechaza un envío declarado para hoy.
APP_TIMEZONE=America/Argentina/Buenos_Aires
APP_LOCALE=es
APP_FALLBACK_LOCALE=es
APP_FAKER_LOCALE=es_AR

APP_MAINTENANCE_DRIVER=file
BCRYPT_ROUNDS=12

# --- Registro ---------------------------------------------------------
# `daily` rota por día. En producción no se registra `debug`: llenaría el
# disco y puede volcar datos de clientes al log.
LOG_CHANNEL=stack
LOG_STACK=daily
LOG_DEPRECATIONS_CHANNEL=null
LOG_LEVEL=warning

# --- Base de datos ----------------------------------------------------
# Los tres valores salen de la sección "Bases de datos" del panel.
DB_CONNECTION=mysql
DB_HOST=localhost
DB_PORT=3306
DB_DATABASE=
DB_USERNAME=
DB_PASSWORD=

# --- Sesión -----------------------------------------------------------
SESSION_DRIVER=database
SESSION_LIFETIME=120
SESSION_ENCRYPT=true
SESSION_PATH=/
SESSION_DOMAIN=null
# Requiere que el SSL del sitio esté ENCENDIDO en el panel. Sin certificado
# el navegador no manda la cookie y el login falla en silencio: se recarga
# el formulario sin ningún mensaje de error.
SESSION_SECURE_COOKIE=true

# --- Cache y colas ----------------------------------------------------
CACHE_STORE=database
QUEUE_CONNECTION=database
BROADCAST_CONNECTION=log

# --- Almacenamiento ---------------------------------------------------
# Los documentos van a un disco privado, fuera de public/. Se sirven por
# ruta firmada con Policy, nunca por URL directa.
FILESYSTEM_DISK=local

# --- Correo -----------------------------------------------------------
# Crear primero la casilla en la sección "Email" del panel.
# MAIL_FROM_ADDRESS tiene que ser la MISMA casilla que MAIL_USERNAME: un
# remitente que no sea una casilla real del dominio autenticado termina
# rechazado o en spam.
MAIL_MAILER=smtp
MAIL_HOST=mail.odessaoil.com
MAIL_PORT=465
MAIL_SCHEME=smtps
MAIL_USERNAME=noreply@odessaoil.com
MAIL_PASSWORD=
MAIL_FROM_ADDRESS="noreply@odessaoil.com"
MAIL_FROM_NAME="${APP_NAME}"

# --- Reglas de negocio ------------------------------------------------
ODESSA_EXPIRY_WARNING_DAYS=60
ODESSA_DEFAULT_VALIDITY_DAYS=365
ODESSA_DOWNLOAD_TTL=5
ODESSA_UPLOAD_MAX_KB=20480
ODESSA_DOCUMENTS_DISK=documents

# --- Ventana de recertificación ----------------------------------------
# El equipo no puede perder la certificación mientras está en el taller:
#     fecha límite de envío = vencimiento − reparación − margen
ODESSA_TURNAROUND_DAYS=30
ODESSA_SAFETY_MARGIN_DAYS=15
ODESSA_RECERT_NOTICE_DAYS=30
ODESSA_RECERT_DIGEST_DAY=1
ODESSA_RECERT_COPY_TO=info@odessaoil.com

# --- Herramientas de desarrollo ---------------------------------------
INERTIA_DEVTOOLS_ENABLED=false

# Vacío en producción: el seeder de demostración no debe correr aquí.
DEMO_PASSWORD=
