# --- Almacenamiento ---------------------------------------------------

# --- Correo -----------------------------------------------------------
# ATENCION: en este servidor la salida SMTP esta BLOQUEADA. Se probaron los
# puertos 587, 465, 25 y 2525 hacia afuera, y 25/587/1025 en local: todos
# cerrados. La configuracion "smtp" de mas abajo NO funciona tal cual esta;
# queda documentada por si habilitan la salida o contratan CloudSMTP.
#
# "sendmail" TAMPOCO sirve: phpinfo muestra la ruta /usr/bin/sendmail, pero
# es solo el valor por defecto de PHP; el binario no esta instalado.
#
# Hasta que haya una via de salida, dejar "log": los avisos quedan escritos
# en storage/logs/ en vez de perderse en silencio.
MAIL_MAILER=log

# Los dos caminos posibles:
#   a) contratar CloudSMTP y completar los datos de abajo con MAIL_MAILER=smtp
#   b) usar un proveedor por API HTTPS (Resend, Postmark, Brevo), que no
#      depende de puertos SMTP y por eso atraviesa este bloqueo. Requiere
#      instalar el paquete en el servidor y agregar registros DNS al dominio.
#
# Puerto 587 con STARTTLS: el esquema va "smtp", no "smtps" (ese es para el
# 465, cifrado desde el saludo). El remitente debe ser la MISMA casilla que
# el usuario autenticado, o el relay rechaza con 535.
MAIL_HOST=COMPLETAR_SERVIDOR_SALIENTE
MAIL_PORT=587
MAIL_SCHEME=smtp
MAIL_USERNAME=noreply@odessaoil.com
MAIL_PASSWORD=COMPLETAR_CLAVE_CORREO
MAIL_FROM_ADDRESS="noreply@odessaoil.com"
MAIL_FROM_NAME="${APP_NAME}"
